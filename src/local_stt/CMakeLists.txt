cmake_minimum_required(VERSION 3.18)
project(ascolto_local_stt LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
if(APPLE)
  set(GGML_METAL ON CACHE BOOL "" FORCE)
else()
  set(GGML_METAL OFF CACHE BOOL "" FORCE)
endif()
get_filename_component(ASCOLTO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include_directories("${ASCOLTO_ROOT}/include")
if(ENGINE STREQUAL "whisper-turbo-q5")
  set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  add_subdirectory("${RUNTIME_SOURCE}" runtime)
  add_library(adapter SHARED whisper.c)
  target_link_libraries(adapter PRIVATE whisper)
elseif(ENGINE STREQUAL "parakeet-v3-q8")
  set(PARAKEET_SHARED OFF CACHE BOOL "" FORCE)
  set(PARAKEET_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(PARAKEET_BUILD_CLI OFF CACHE BOOL "" FORCE)
  set(PARAKEET_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  set(PARAKEET_WITH_CED OFF CACHE BOOL "" FORCE)
  set(PARAKEET_WITH_VOICEDETECT OFF CACHE BOOL "" FORCE)
  set(PARAKEET_GGML_METAL ${APPLE} CACHE BOOL "" FORCE)
  add_subdirectory("${RUNTIME_SOURCE}" runtime)
  # Upstream uses standalone source-root includes; provide them when embedded.
  target_include_directories(dr_wav_impl PRIVATE "${RUNTIME_SOURCE}/third_party")
  target_include_directories(parakeet PRIVATE "${RUNTIME_SOURCE}/third_party")
  add_library(adapter SHARED parakeet.cpp)
  target_include_directories(adapter PRIVATE "${RUNTIME_SOURCE}/src")
  target_link_libraries(adapter PRIVATE parakeet)
elseif(ENGINE STREQUAL "qwen3-asr-1.7b")
  set(QWEN_SOURCES qwen_asr.c qwen_asr_kernels.c qwen_asr_kernels_generic.c qwen_asr_kernels_neon.c
      qwen_asr_kernels_avx.c qwen_asr_audio.c qwen_asr_encoder.c qwen_asr_decoder.c qwen_asr_tokenizer.c qwen_asr_safetensors.c)
  list(TRANSFORM QWEN_SOURCES PREPEND "${RUNTIME_SOURCE}/")
  add_library(adapter SHARED qwen.c ${QWEN_SOURCES})
  target_include_directories(adapter PRIVATE "${RUNTIME_SOURCE}")
  target_compile_definitions(adapter PRIVATE USE_BLAS)
  if(APPLE)
    target_compile_definitions(adapter PRIVATE ACCELERATE_NEW_LAPACK)
    target_link_libraries(adapter PRIVATE "-framework Accelerate")
  else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(OPENBLAS REQUIRED IMPORTED_TARGET openblas)
    target_compile_definitions(adapter PRIVATE USE_OPENBLAS)
    target_link_libraries(adapter PRIVATE PkgConfig::OPENBLAS m pthread)
  endif()
else()
  message(FATAL_ERROR "Unknown ENGINE")
endif()
set_target_properties(adapter PROPERTIES PREFIX "" OUTPUT_NAME "${ENGINE}" LIBRARY_OUTPUT_DIRECTORY "${OUTPUT_DIRECTORY}")
# Static dependencies occasionally declare default visibility explicitly. Export only the adapter ABI.
if(APPLE)
  target_link_options(adapter PRIVATE "-Wl,-exported_symbol,_ascolto_stt_get_api")
else()
  file(WRITE "${CMAKE_BINARY_DIR}/exports.map" "{ global: ascolto_stt_get_api; local: *; };\n")
  target_link_options(adapter PRIVATE "-Wl,--version-script=${CMAKE_BINARY_DIR}/exports.map" "-Wl,-Bsymbolic")
endif()
